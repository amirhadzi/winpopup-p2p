cmake_minimum_required(VERSION 3.20)
project(WinPopupP2P VERSION 0.2.0 LANGUAGES CXX RC)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Build with MSVC for Windows x64")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(POPUP_DEPS "${CMAKE_CURRENT_SOURCE_DIR}/../deps/install" CACHE PATH "Directory containing include/ and lib/ for the pinned native dependencies")
add_library(popup_core STATIC src/core.cpp)
target_include_directories(popup_core PUBLIC src "${POPUP_DEPS}/include")
target_compile_definitions(popup_core PUBLIC UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00 SODIUM_STATIC __PTW32_STATIC_LIB)
target_compile_options(popup_core PUBLIC /utf-8 /W4 /permissive- /guard:cf)
target_link_libraries(popup_core PUBLIC
  "${POPUP_DEPS}/lib/toxcore.lib"
  "${POPUP_DEPS}/lib/libsodium.lib"
  "${POPUP_DEPS}/lib/pthreadVC3.lib"
  ws2_32 iphlpapi advapi32 bcrypt)
add_executable(WinPopup WIN32 src/main.cpp resources/app.rc)
target_include_directories(WinPopup PRIVATE resources)
target_link_libraries(WinPopup PRIVATE popup_core comctl32 shell32 uxtheme dwmapi msimg32)
target_link_options(WinPopup PRIVATE /MANIFEST:NO /DYNAMICBASE /NXCOMPAT /HIGHENTROPYVA /guard:cf)
add_executable(core_tests tests/core_tests.cpp)
target_link_libraries(core_tests PRIVATE popup_core)
add_executable(network_probe tests/network_probe.cpp)
target_link_libraries(network_probe PRIVATE popup_core)
add_executable(ui_render tests/ui_render.cpp resources/app.rc)
target_include_directories(ui_render PRIVATE resources)
target_link_libraries(ui_render PRIVATE popup_core comctl32 shell32 uxtheme dwmapi msimg32)
target_link_options(ui_render PRIVATE /MANIFEST:NO)
add_executable(ui_behavior_tests tests/ui_behavior_tests.cpp)
target_link_libraries(ui_behavior_tests PRIVATE popup_core comctl32 shell32 uxtheme dwmapi msimg32)
enable_testing()
add_test(NAME core_integration COMMAND core_tests)
add_test(NAME ui_behavior COMMAND ui_behavior_tests)
set_tests_properties(core_integration PROPERTIES TIMEOUT 240)
set_tests_properties(ui_behavior PROPERTIES TIMEOUT 30)
install(TARGETS WinPopup RUNTIME DESTINATION .)
